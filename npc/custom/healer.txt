//===== rAthena Script =======================================
//= Malangdo Corp Healer
//= Quest NPC and Healing Service unlock
//============================================================

iz_ac01,93,156,6	script	Healer	437,{
	if (checkquest(90073) == 1) {
		mes "[Cindy]";
		mes "Hi again, " + strcharinfo(0) + "!";
		next;

		mes "[Cindy]";
		mes "In case you've forgotten, I'm Cindy. Whenever you need me, I'll restore you to full health and send you on your way with a fresh set of buffs.";
		mes "No zeny. No waiting. Just ask.";
		next;

		mes "[Cindy]";
		mes "I can also <b>cleanse your debuffs, identify your items, and repair your equipment.</b>";
		mes "Think of me as your one-stop shop for getting back into the fight!";
		next;

		mes "You've recovered ^i[44599] 1x ^0000FF<b>Memory Shard!</b>^000000";
		mes "You gained ^FFA500some experience back^000000.";

		completequest 90073;
		setquest 90074;
		end;
	}

	mes "[Cindy]";
	mes "I'm here whenever you need me.";
	end;

OnInit:
	setunittitle(getnpcid(0), "[Malangdo Corp]");
	end;
}
