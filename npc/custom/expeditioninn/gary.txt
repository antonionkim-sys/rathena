prtinn,59,52,5	script	Scientist Gary	402,{
	mes "[Scientist Gary]";
	mes "I have Good and Evil gears. In return, you need to give me the ticket.";
	mes "You can merge excessive gears into special tickets in the Item Workshop.";
	mes "Using this special essence, you can buy another crown or weapon.";
	next;

	switch (select("Crown Coupon exchange:Weapon Essence exchange:Cancel")) {
	case 1:
		mes "[Scientist Gary]";
		mes "Choose a crown that you need.";
		mes "You can get ^i[44187] from the ^0000FFLarge Recycle Machine^000000 inside the Item Workshop.";
		close2;
		callshop "Gary_GoodEvil_Crowns", 1;
		end;
	case 2:
		mes "[Scientist Gary]";
		mes "Choose a weapon that you need.";
		mes "You can get ^i[44188] from the ^0000FFLarge Recycle Machine^000000 inside the Item Workshop.";
		close2;
		callshop "Gary_GoodEvil_Weapons", 1;
		end;
	case 3:
		close;
	}

OnInit:
	setunittitle(getnpcid(0), "[Good and Evil Gears]");
	end;
}
