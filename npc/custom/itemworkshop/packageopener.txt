//===== rAthena Script =======================================
//= Fast-Hand Gaot - Package Opener
//============================================================
//= Fee: 3,000 Zeny per package.
//= Maximum batch: 100 packages.
//= Capacity checks run before each opening.
//============================================================

prontera,187,209,3	script	Fast-Hand Gaot	804,{
	mes "[Fast-Hand Gaot]";
	mes "I can help you quickly unbox the following item packages if you have them.";
	mes "^FF0000Note: Ensure you're not overweight and have plenty of inventory space.^000000";
	mes "Each unbox costs 3,000 Zeny.";
	next;

	.@choice = select(
		"Old Blue Box (" + countitem(603) + ")",
		"Old Purple Box (" + countitem(617) + ")",
		"Old Card Album (" + countitem(616) + ")",
		"Gift Box (" + countitem(644) + ")",
		"Hero Gift Box (" + countitem(101750) + ")",
		"Cancel"
	);

	switch (.@choice) {
	case 1:
		.@box = 603;
		.@group = IG_BlueBox;
		break;

	case 2:
		.@box = 617;
		.@group = IG_VioletBox;
		break;

	case 3:
		.@box = 616;
		.@group = IG_CardAlbum;
		break;

	case 4:
		.@box = 644;
		.@group = IG_GiftBox;
		break;

	case 5:
		.@box = 101750;
		.@group = IG_HEROSRIA_GIFT;
		break;

	default:
		close;
	}

	// --------------------------------------------------
	// Check owned packages.
	// --------------------------------------------------
	.@owned = countitem(.@box);

	if (.@owned < 1) {
		mes "[Fast-Hand Gaot]";
		mes "You don't have any " + mesitemlink(.@box) + ".";
		close;
	}

	.@limit = min(.@owned,100);

	mes "[Fast-Hand Gaot]";
	mes "How many would you like me to open?";
	mes "You can open between 1 and " + .@limit + " packages in this batch.";
	next;

	input .@quantity,1,.@limit;

	.@cost = .@quantity * 3000;

	mes "[Fast-Hand Gaot]";
	mes "Packages: " + .@quantity + "x " + mesitemlink(.@box);
	mes "Total fee: " + .@cost + " Zeny.";
	next;

	if (select("Open packages:Cancel") == 2)
		close;

	// --------------------------------------------------
	// Recheck requirements after confirmation.
	// --------------------------------------------------
	if (countitem(.@box) < .@quantity) {
		mes "[Fast-Hand Gaot]";
		mes "You no longer have enough packages.";
		close;
	}

	if (Zeny < .@cost) {
		mes "[Fast-Hand Gaot]";
		mes "You need " + .@cost + " Zeny for this batch.";
		close;
	}

	// --------------------------------------------------
	// Open packages individually.
	// Charge only for packages processed.
	// --------------------------------------------------
	.@opened = 0;

	for (.@i = 0; .@i < .@quantity; .@i++) {
		// Stay below 70% carrying weight.
		if (Weight * 100 >= MaxWeight * 70)
			break;

		getinventorylist;

		// Keep at least 10 inventory slots free.
		if (@inventorylist_count > .InventoryLimit - 10)
			break;

		if (countitem(.@box) < 1 || Zeny < 3000)
			break;

		delitem .@box,1;
		Zeny -= 3000;
		getgroupitem .@group;

		.@opened++;
	}

	// --------------------------------------------------
	// Results.
	// --------------------------------------------------
	mes "[Fast-Hand Gaot]";

	if (.@opened == 0) {
		mes "I couldn't start opening your packages.";
		mes "Please stay below 70% carrying weight and keep at least 10 inventory slots free.";
		mes "No packages or Zeny were consumed.";
		close;
	}

	mes "Opened " + .@opened + " package(s).";
	mes "Total fee paid: " + (.@opened * 3000) + " Zeny.";

	if (.@opened < .@quantity) {
		mes "I stopped before completing the batch.";
		mes "Please check your inventory space, carrying weight, remaining packages, and Zeny.";
		mes "Unopened packages were not charged.";
	}

	close;

OnInit:
	setunittitle(getnpcid(0), "[Package Opener]");

	// Match this to your server's actual inventory slot limit.
	.InventoryLimit = 100;
	end;
}
