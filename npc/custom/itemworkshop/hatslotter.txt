//= Slot Artisan - Hat Slotter
//============================================================
//= 6396: 100% success.
//= 25041: 25% success.
//= Failure consumes only the catalyst.
//= Success gives a fresh slotted item.
//= No Zeny fee.
//= Both catalysts are available for every listed hat.
//============================================================

	mes "[Slot Artisan]";
	mes "If you have a ^i[6396] or ^i[25041], I can use it to add a card slot to your headgear.";

		mes "I can add a card slot to supported headgear.";
		mes "For that, I need special items to process.";
		next;

		mes "[Slot Artisan]";
		mes "Using ^i[6396], the success rate is 100%.";
		mes "Using ^i[25041], the success rate is 25%.";
		mes "On failure, only ^i[25041] is lost.";
		mes "Your hat stays unchanged.";
		next;

		mes "[Slot Artisan]";
		mes "You can buy ^i[25041] from the Department Store.";
		mes "As for ^i[6396], it comes from Free Egg Scrolls.";
		mes "It totally depends on your luck.";
		close;

	case 2:
		break;

	default:
		close;
	}
	mes "";
	mes "So, which headgear do you want me to work on?";
	next;

	.@choice = select(.menu$ + ":Cancel") - 1;

	if (.@choice >= getarraysize(.base))
		close;

	.@base = .base[.@choice];
	.@result = .result[.@choice];

	mes "[Slot Artisan]";
	mes "Would you like to use ^i[6396] or ^i[25041]?";
	next;

	switch(select(
		"Spiritual Auger",
		"Soul Plunger",
		"Cancel"
	)) {
	case 1:
		.@catalyst = 6396;
		.@chance = 100;
		break;

	case 2:
		.@catalyst = 25041;
		.@chance = 25;
		break;

	default:
		close;
	}

	mes "[Slot Artisan]";

	if (.@catalyst == 25041) {
		mes "If slot creation fails, you will lose " + mesitemlink(25041) + ".";
		mes "Your hat will NOT be destroyed on failure.";
	}

	mes "On success, the selected hat becomes a fresh slotted version.";
	mes "^FF0000Refines, enchants, random options, and other item-specific data will not be transferred.^000000";
	mes "Please unequip the hat first.";
	mes "Bound and rental hats are not accepted.";
	mes "Success rate: " + .@chance + "%";
	next;

	if (select(
		"^0000FFAsk for slot creation^000000",
		"^FF0000Try next time^000000"
	) == 2)
		close;

	// --------------------------------------------------
	// Check source headgear quantity.
	// Missing and duplicate items have separate messages.
	// --------------------------------------------------
	.@hat_count = countitem(.@base);

	if (.@hat_count < 1) {
		mes "[Slot Artisan]";
		mes "You don't have the required unslotted headgear:";
		mes mesitemlink(.@base);
		close;
	}

	if (.@hat_count > 1) {
		mes "[Slot Artisan]";
		mes "You have more than one copy of the chosen unslotted headgear.";
		mes "Please store the extra copies and carry only one.";
		close;
	}

	// --------------------------------------------------
	// Check catalyst.
	// --------------------------------------------------
	if (countitem(.@catalyst) < 1) {
		mes "[Slot Artisan]";
		mes "You don't have the required catalyst:";
		mes mesitemlink(.@catalyst);
		close;
	}

	// --------------------------------------------------
	// Find the exact source inventory slot.
	// --------------------------------------------------
	getinventorylist;
	.@idx = -1;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@base)
			continue;

		if (@inventorylist_equip[.@i]) {
			mes "[Slot Artisan]";
			mes "Please unequip the selected headgear first.";
			close;
		}

		if (@inventorylist_bound[.@i]) {
			mes "[Slot Artisan]";
			mes "Bound headgear cannot be processed.";
			close;
		}

		if (@inventorylist_expire[.@i]) {
			mes "[Slot Artisan]";
			mes "Rental headgear cannot be processed.";
			close;
		}

		if (!@inventorylist_identify[.@i]) {
			mes "[Slot Artisan]";
			mes "Please identify the selected headgear first.";
			close;
		}

		.@idx = @inventorylist_idx[.@i];
		break;
	}

	if (.@idx < 0) {
		mes "[Slot Artisan]";
		mes "I couldn't locate the selected headgear in your inventory.";
		close;
	}

	// --------------------------------------------------
	// Conservative capacity check before consuming items.
	// --------------------------------------------------
	if (!checkweight(.@result,1)) {
		mes "[Slot Artisan]";
		mes "Please free inventory space and reduce your carried weight.";
		close;
	}

	// --------------------------------------------------
	// Consume catalyst and perform the success roll.
	// Failure leaves the original hat untouched.
	// --------------------------------------------------
	delitem .@catalyst,1;

	if (rand(1,100) > .@chance) {
		mes "[Slot Artisan]";
		mes "Slot creation failed.";
		mes "The Soul Plunger was consumed.";
		mes "Your headgear is unchanged.";
		close;
	}

	// --------------------------------------------------
	// Success: replace the exact source hat.
	// --------------------------------------------------
	delitemidx .@idx,1;
	getitem .@result,1;

	mes "[Slot Artisan]";
	mes "Success! Here is your " + mesitemlink(.@result) + ".";
	close;
