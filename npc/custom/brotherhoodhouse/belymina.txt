	if (checkquest(90006) == -1) {

		mes "Yo, "+strcharinfo(0)+"! I've been expecting you!";

		next;

		mes "[Belymina]";
		mes "I used to be part of the Eden Group, researching highly advanced tactical weapons. Eventually, I joined the Bromance Union to guide novices and aspiring adventurers.";
		next;

		mes "[Belymina]";
		mes "I wanted to help them survive their first real battles, so I developed weapons designed to help them grow into capable fighters.";
		next;

		mes "[Belymina]";
		mes "After many trials and years of research, I finally figured out how to improve the tactical weapons supplied by the Eden Group.";

		mes "[Belymina]";
		mes "The Eden Group was established in response to the growing threats facing Midgard, bringing together skilled fighters to prepare for the catastrophe on the horizon.";

		completequest 90006;

		close2;
		cutin "",255;
		end;
	}

	// Quest 90009 is active:

	mes "[Belymina]";
	mes "Have you gotten those tubes yet?";
	next;

		close2;
		cutin "",255;
		end;
	}

	delitem 7325,20;

	mes "[Belymina]";
	mes "Excellent! These supplies will help keep our weapon research running.";
	next;

	mes "[Belymina]";
	mes "Most people think weapons are made from metal, but that's only half the story.";
	next;

	mes "[Belymina]";
	mes "A weapon is only as good as the person using it. The strongest blade in the world is useless in the hands of someone who hasn't learned how to wield it.";
	next;

	mes "[Belymina]";
	mes "You've got potential, but you're not ready for the work I have waiting for you.";
	next;

	mes "[Belymina]";
	mes "As promised, here is your reward.";
	
	getitem 1000322,20;
	getexp 20693,5000;
	completequest 90009;	
	setquest 90010;
	
	next;
	
	mes "[Belymina]";
	mes "That's all I can offer you for now. Come back when you've reached third job.";

	close2;
	cutin "",255;
	end;

	completequest 90010;

	close2;
	cutin "",255;
	end;

			"Buy a Booster Weapon.",
			"Refine & Random Options a Booster Weapon.",
			"Enchant a Booster Weapon.",

			mes "[Belymina]";
			mes "Take a look.";

			cutin "",255;
