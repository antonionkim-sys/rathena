iz_ac01	mapflag	loadevent

iz_ac01,96,31,5	script	Secretary Snow	864,{
	cutin "secretary_snow", 2;

	// =====================================================
	// CONFIGURATION
	// =====================================================
	.@TOKEN_ID = 25051;
	.@MAX_REPUTATION = 5000;

	.@token_icon$ = "^i[" + .@TOKEN_ID + "]";

	.@eden_runes$ = "^FFA500Eden Runes^000000";
	.@prontera$ = "^0000FFProntera^000000";


	// =====================================================
	// NOT READY YET
	// =====================================================
	if (isbegin_quest(90003) == 0) {
		mes "[Secretary Snow]";
		mes "Heya! Are you new? Do you have an invitation?";
		next;

		mes "[Secretary Snow]";
		mes "You're not expected here yet.";

		cutin "", 255;
		close;
	}


	// =====================================================
	// INITIAL INTRODUCTION
	// QUEST 90003
	// =====================================================
	if (isbegin_quest(90003) == 1 && checkquest(90003) != 2) {
		doevent(strnpcinfo(3) + "::OnTalk");
		end;
	}


	// =====================================================
	// QUEST 90004 ACTIVE
	// =====================================================
	if (checkquest(90003) == 2 && checkquest(90004) == 1) {
		mes "[Secretary Snow]";
		mes "Before you head out, please go speak with our Kafra employee at the front desk.";
		mes "She will introduce you to everyone stationed here and get you fully acquainted with the facility.";

		cutin "", 255;
		close;
	}


	// =====================================================
	// MEMORY SHARD QUEST
	// QUEST 90071
	// =====================================================
	if (checkquest(90071) == 1) {

		mes "[Secretary Snow]";
		mes "There's something I just remembered...You and Sake.";
		mes "You two were practically inseparable back then.";
		mes "Wherever there was trouble, the two of you were already in the middle of it.";
		next;

		mes "[Secretary Snow]";
		mes "I still remember watching you come back from one battle after another,";
		mes "covered in dirt, blood, and somehow still laughing and drinking about it.";
		next;

		mes "[Secretary Snow]";
		mes "Eventually, Sake took over the Brotherhood and started running everything himself.";
		mes "";
		mes "That's when I started noticing how quickly he began to age...";
		next;

		mes "[Secretary Snow]";
		mes "He'd stand here for hours, fussing over Union paperwork,";
		mes "moving papers around, sighing at them...";
		mes "Although, I never once saw him actually file any of it.";
		next;

		mes "[Secretary Snow]";
		mes "He'd complain that the Brotherhood was too much work,";
		mes "then spend the entire night working anyway.";
		next;

		mes "[Secretary Snow]";
		mes "I suppose that's what it cost him.";
		mes "He gave so much of himself to this place that it seemed to age him years at a time.";
		next;

		mes "[Secretary Snow]";
		mes "But knowing Sake...";
		mes "I'm sure he'd complain about that too.";
		next;

		mes "[Secretary Snow]";
		mes "You've recovered ^i[44599] 1x ^0000FF<b>Memory Shard!</b>^000000";
		mes "You gained ^FFA500some experience back^000000.";

		completequest 90071;
		setquest 90072;

		close;
	}


	// =====================================================
	// RETURN VISIT AFTER QUEST 90004
	// =====================================================
	if (checkquest(90004) == 2) {

		mes "[Secretary Snow]";
		mes "Hiya, " + strcharinfo(0) + "~!";
		mes "Welcome back to the house of Brotherhood.";
		mes "How can I assist you today? ^^";
		next;

		switch (select(
			"What should I do next?",
			"Boost my Reputation!",
			"Cancel"
		)) {

			// =================================================
			// WHAT SHOULD I DO NEXT?
			// =================================================
			case 1:
				mes "[Secretary Snow]";
				mes "Keep these core directives in mind:";
				mes "^0000FF- Slay monsters^000000 to salvage Union Tokens.";
				mes "^0000FF- Invest tokens^000000 into the Brotherhood House Reputation.";
				mes "^0000FF- Channel tokens^000000 to awaken your " + .@eden_runes$ + ".";
				next;

				mes "[Secretary Snow]";
				mes "Maxing out your Reputation is vital.";
				mes "Our quartermasters will only unlock their advanced supply caches once you've proven your absolute loyalty to the House.";

				cutin "", 255;
				close;


			// =================================================
			// BOOST REPUTATION
			// =================================================
			case 2:

				while (1) {
					mes "[Secretary Snow]";
					mes "Oooh! Want to help our Brotherhood House grow?";
					mes "Every " + .@token_icon$ + " you donate gives us 1 Reputation point!";
					mes "Reach 1,000 points and we'll level up together!";
					mes " ";
					mes "Current <B>Reputation</B>: ^0000FF"
						+ RepPointsBrotherhood
						+ "^000000 / "
						+ .@MAX_REPUTATION;
					next;

					.@rep_choice = select(
						"Donate 1 Union Token",
						"Donate 10 Union Tokens",
						"Donate 100 Union Tokens",
						"Tell me about Reputation",
						"Back"
					);

					if (.@rep_choice == 5)
						break;


					// =============================================
					// REPUTATION INFORMATION
					// =============================================
					if (.@rep_choice == 4) {
						mes "[Secretary Snow]";
						mes "Here's a cute little guide to check your Reputation status!";
						next;

						cutin "reputation_guide", 4;

						mes "[Secretary Snow]";
						mes "Reputation is like making friends with different groups!";
						mes "The more they trust you, the more special things they'll share!";
						next;

						mes "[Secretary Snow]";
						mes "Different groups like different gifts-we love Etel Dust!";
						mes "Other places might prefer different materials~";
						next;

						mes "[Secretary Snow]";
						mes "When you build reputation with us, we'll show you extra special items in our shop!";
						mes "Isn't that exciting? ^^";
						next;

						cutin "secretary_snow", 2;
						close;
					}


					// =============================================
					// DONATION AMOUNT
					// =============================================
					.@donation_amount = 1;

					if (.@rep_choice == 2)
						.@donation_amount = 10;

					if (.@rep_choice == 3)
						.@donation_amount = 100;


					// =============================================
					// MAX REPUTATION
					// =============================================
					if (RepPointsBrotherhood >= .@MAX_REPUTATION) {
						mes "[Secretary Snow]";
						mes "Your Brotherhood Reputation is already at the current maximum of "
							+ .@MAX_REPUTATION
							+ " points.";
						continue;
					}


					// =============================================
					// PREVENT OVER-CAPPING
					// =============================================
					.@reputation_remaining =
						.@MAX_REPUTATION - RepPointsBrotherhood;

					if (.@donation_amount > .@reputation_remaining)
						.@donation_amount = .@reputation_remaining;


					// =============================================
					// TOKEN CHECK
					// =============================================
					if (countitem(.@TOKEN_ID) < .@donation_amount) {
						mes "[Secretary Snow]";
						mes "You don't have enough Union Tokens.";
						mes "You need ^0000FF"
							+ .@donation_amount
							+ "^000000, but you only have ^0000FF"
							+ countitem(.@TOKEN_ID)
							+ "^000000.";
						continue;
					}


					// =============================================
					// DONATE
					// =============================================
					delitem .@TOKEN_ID, .@donation_amount;
					RepPointsBrotherhood += .@donation_amount;

					mes "[Secretary Snow]";
					mes "Thank you for your donation!";
					mes "Your Brotherhood Reputation increased by ^0000FF"
						+ .@donation_amount
						+ "^000000.";
					mes "Current <B>Reputation</B>: ^0000FF"
						+ RepPointsBrotherhood
						+ "^000000 / "
						+ .@MAX_REPUTATION;
					next;


					// =============================================
					// MAX REPUTATION MESSAGE
					// =============================================
					if (RepPointsBrotherhood >= .@MAX_REPUTATION) {
						mes "[Secretary Snow]";
						mes "Amazing! We've reached 1,000 Reputation points!";
						mes "The Brotherhood House has leveled up together with you!";
					}
				}

				cutin "", 255;
				close;


			// =================================================
			// CANCEL
			// =================================================
			case 3:
				cutin "", 255;
				close;
		}
	}


	// =====================================================
	// FALLBACK
	// =====================================================
	mes "[Secretary Snow]";
	mes "Something seems out of order with your records.";
	mes "Speak with the Valkyrie or try again later.";

	cutin "", 255;
	close;


	// =====================================================
	// AUTOMATIC SNOW INTRO
	// Triggered by iz_ac01 loadevent
	// =====================================================
OnPCLoadMapEvent:
	if (strcharinfo(3) != "iz_ac01")
		end;

	if (isbegin_quest(90003) == 1 && checkquest(90003) != 2) {
		doevent(strnpcinfo(3) + "::OnTalk");
	}

	end;


	// =====================================================
	// SNOW INTRO DIALOGUE
	// Shared by click + automatic map entry
	// =====================================================
OnTalk:
	.@TOKEN_ID = 25051;
	.@token_icon$ = "^i[" + .@TOKEN_ID + "]";
	.@eden_runes$ = "^FFA500Eden Runes^000000";
	.@prontera$ = "^0000FFProntera^000000";

	cutin "secretary_snow", 2;

	mes "[Secretary Snow]";
	mes "Greetings. I've been expecting you.";
	mes "^0000FFValkyrie Sigrun^000000 reached out to me and relayed everyth..??";
	next;

	mes "[Secretary Snow]";
	mes "...?!";
	mes strcharinfo(0) + "... Is that really you?";
	next;

	mes "[Secretary Snow]";
	mes "Where have you been this whole time??";
	mes "The last thing I remember is you fighting ^0000FFSatan Morroc^000000...";
	mes "And then... no one ever heard from you again.";
	next;

	mes "[Secretary Snow]";
	mes "And it looks like you haven't aged a bit...";
	next;

	mes "[Secretary Snow]";
	mes "I heard you've lost your memories...";
	mes "I'm so sorry...";
	mes "But perhaps... we can start again...";
	next;

	mes "[Secretary Snow]";
	mes "I am Snow.";
	mes "I am now the House Manager of the Brotherhood, and I keep watch over this clan's very first door.";
	next;

	mes "[Secretary Snow]";
	mes "Although... when we first met, I was just a novice.";
	mes "A lot has changed since then.";
	next;

	mes "[Secretary Snow]";
	mes "And now, somehow...";
	mes "you're standing here in front of me again.";
	mes "After all these years...";
	next;

	completequest 90003;
	setquest 90004;

	getitem .@TOKEN_ID, 1;

	dispbottom "System: You have officially joined the Brotherhood Alliance.";
	dispbottom "System: You have received 1x " + getitemname(.@TOKEN_ID) + ".";

	showevent 1, 0;

	sc_start SC_BLESSING_OF_THE_BROTHERHOOD, 600000, 10;
	specialeffect EF_WARP;

	mes "[Secretary Snow]";
	mes "There. You are now recognized by the union again.";
	mes "Take this " + .@token_icon$ + ". It is a token of our Brotherhood.";
	next;

	mes "[Secretary Snow]";
	mes "Consider it a sacred symbol of your membership.";
	mes "As an active vanguard, any monster you slay across the world now carries a chance to yield a similar token recovered from those who have fallen.";
	next;

	mes "[Secretary Snow]";
	mes "These tokens are the lifeblood of our resistance.";
	mes "You can channel them to fuel your " + .@eden_runes$ + " for immense raw power,";
	mes "or invest them directly into the Brotherhood House Reputation.";
	next;

	mes "[Secretary Snow]";
	mes "Every " + .@token_icon$ + " you donate gains one <B>Reputation</B> point,";
	mes "or you can feed it to the Eden runes at the Rune Tablet.";
	next;

	mes "[Secretary Snow]";
	mes "Maximize that reputation score!";
	mes "The higher your standing, the more exclusive and powerful equipment our restricted shops will open up to you.";
	next;

	mes "[Secretary Snow]";
	mes "As a welcome gift, I have also permanently infused your essence with a protective union blessing,";
	mes "granting you a small increase to all core attributes.";
	next;

	mes "[Secretary Snow]";
	mes "Most of our active members are now currently scattered across the world out on the frontlines,";
	mes "though a large contingent of them presently reside within the capital city of " + .@prontera$ + ".";
	next;

	mes "[Secretary Snow]";
	mes "I'm sure you will meet some familiar faces again.";
	next;

	mes "[Secretary Snow]";
	mes "This Academy serves as our main stronghold and sanctuary.";
	mes "You are welcome here at any time, day or night.";
	next;

	mes "[Secretary Snow]";
	mes "Before you head out, please go speak with our Kafra employee at the front desk.";
	mes "She will introduce you to everyone stationed here and get you fully acquainted with the facility.";
	next;

	mes "[Secretary Snow]";
	mes "And " + strcharinfo(0) + "...";
	mes "If you truly want your memories back...";
	mes "don't give up.";
	next;

	mes "[Secretary Snow]";
	mes "Welcome home.";

	close2;
	cutin "", 255;
	navigateto "iz_ac01", 96, 41;
	end;
