	mes "[Nyarom]";
	mes "Welcome, welcome.";
	mes "This is the Random Options shop.";

	next;

	switch(select(
		"Buy Special Item Random Options (Silvervine)",
		"Buy Special Item Random Options (Zeny)",

		"Buy Instance Random Options",
		"Buy Shadow Gear Random Options",

		callshop "Nyarom_Silvervine",1;
		end;

	case 2:
		mes "[Nyarom]";
		mes "There are special items for you.";
		close2;
		callshop "Nyarom_Zeny",1;
		end;

	case 3:
		mes "[Nyarom]";
		mes "We have prepared the Bravery and Wisdom of Great Hero for our special customer.";
		close2;
		callshop "Nyarom_Hero",1;
		end;

	case 4:
		mes "[Nyarom]";
		mes "These ones are special catalysts for items of instances.";
		mes "The materials needed to exchange can be farmed from instances.";
		close2;
		callshop "Nyarom_Instance",1;
		end;

	case 5:
		mes "[Nyarom]";
		mes "These ones are special catalysts for Shadow equipment.";
		close2;
		callshop "Nyarom_Shadow",1;
		end;

	case 6:
		mes "[Nyarom]";
		mes "If you want to make your dungeon equipment more powerful, get the random options!";
		close2;
		callshop "Nyarom_Dungeon",1;
		end;

	default:
		close;
	}
