//============================================================
// Equipment list helper.
// Arguments are alternating base/result item IDs.
// Displays five pairs per page.
//============================================================
L_List:
	mes "[Gigantic Device]";
	mes "List of items";

	for (.@i = 0; .@i < getargcount(); .@i += 2) {
		mes "^i[" + getarg(.@i) + "] -> ^i[" + getarg(.@i + 1) + "]";
	}

	next;
	return;

	setunittitle(getnpcid(0), "[Item Reform]");
	end;
